255        """
256        # linearized electron density  N_e*l calculation 
257        Args:
258            None
259        Returns:

261        """
262        try:
263            Time = np.linspace(0,1e-6*(50000-1),50000)
264
265            interferometry = MW_interferometry()
266            Divider = 1
267            DiscFreq = 1e7 / Divider
268            IntermFreq = 1e6
269            REF = self.hdf5_file['interferometry']['interf_detector2']['data'][:]
270            RF = self.hdf5_file['interferometry']['interf_detector1']['data'][:]
271
272            # Apply bandisth filters
273            REF_F = interferometry.FourierCut(REF, DiscFreq, 10*1e3, IntermFreq)          #10*1e3 - filter FWHM
274            RF_F = interferometry.FourierCut(RF, DiscFreq, 10*1e3, IntermFreq)
275            time_density = self.hdf5_file['interferometry']['interf_detector2']['time'][:]

280
281            # Calculate density
282            phase = -(ph_RF - ph_REF)
283            phase = phase - np.mean(phase[0:100]) # Remove initial phase
284            density = abs(interferometry.PhaseToDensity(phase))
285            n_el = np.interp(Time,time_density*1000,density)
286            return time_density, abs(density)
