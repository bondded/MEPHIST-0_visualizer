 79    def _format_speed(self, bytes_per_second: float) -> str:
 80        """
 81        Форматирует скорость в человеко-читаемый вид.
 82        
 83        Args:
 84            bytes_per_second (float): Скорость в байтах в секунду
 85            
 86        Returns:
 87            str: Отформатированная строка скорости
 88        """
 89        if bytes_per_second >= 1024 ** 3:  # GB/s
 90            return f"{bytes_per_second / (1024 ** 3):.2f} GB/s"
 91        elif bytes_per_second >= 1024 ** 2:  # MB/s
 92            return f"{bytes_per_second / (1024 ** 2):.2f} MB/s"
 93        elif bytes_per_second >= 1024:  # KB/s
 94            return f"{bytes_per_second / 1024:.2f} KB/s"
 95        else:  # B/s
 96            return f"{bytes_per_second:.0f} B/s"

 99        """
100        Форматирует время в человеко-читаемый вид.
101        
102        Args:
103            seconds (float): Время в секундах
104            
105        Returns:
106            str: Отформатированная строка времени
107        """
108        if seconds < 1:
109            return f"{seconds * 1000:.0f} мс"
110        elif seconds < 60:
111            return f"{seconds:.1f} с"
112        elif seconds < 3600:
113            minutes = int(seconds // 60)
114            secs = seconds % 60
115            return f"{minutes} мин {secs:.0f} с"
116        else:
117            hours = int(seconds // 3600)
118            minutes = int((seconds % 3600) // 60)
119            secs = seconds % 60
120            return f"{hours} ч {minutes} мин {secs:.0f} с"

262    def _parse_cache_size(self, size_str: str) -> int:
263        """
264        Парсит строку с размером кэша (например, "500 MB", "1GB", "100KB").
265        
266        Args:
267            size_str (str): Строка с размером
268            
269        Returns:
270            int: Размер в байтах
271        """
272        if not size_str or size_str == "0":
273            return 0
274        
275        size_str = size_str.lower().replace(" ", "")
276        
277        # Регулярное выражение для поиска числа и единицы измерения
278        pattern = r'^(\d+(?:\.\d+)?)([kmgt]?b)$'
279        match = re.match(pattern, size_str)
280        
281        if not match:
282            error_message = f"Неверный формат размера кэша в config.yaml: {size_str}. Используйте формат типа '500MB', '1GB', '100KB'"
283            self.logger.error(error_message)
284            sys.exit(1)  
285        
286        size_value = float(match.group(1))
287        unit = match.group(2)
288        
289        units = {
290            'b': 1,
291            'kb': 1024,
292            'mb': 1024 ** 2,
293            'gb': 1024 ** 3,
294            'tb': 1024 ** 4
295        }
296        
297        if unit not in units:
298            error_message = f"Неизвестная единица измерения в config.yaml: {unit}. Используйте B, KB, MB, GB, TB"
299            self.logger.error(error_message)
300            sys.exit(1)       
301        
302        return int(size_value * units[unit])

304    def _get_cached_files_info(self) -> List[Tuple[Path, int, float]]:
305        """
306        Получает информацию о файлах в кэше.
307        
308        Returns:
309            List[Tuple[Path, int, float]]: Список кортежей (путь, размер, время модификации)
310        """
311        cache_path = Path(self.cache_dir)
312        if not cache_path.exists():
313            return []
314        
315        files_info = []
316        for cache_file in cache_path.glob("*.nxs"):
317            if cache_file.exists():
318                stat = cache_file.stat()
319                files_info.append((cache_file, stat.st_size, stat.st_mtime))
320        
321        # Сортируем по времени модификации (старые файлы первыми)
322        files_info.sort(key=lambda x: x[2])
323        return files_info

325    def _cleanup_cache_if_needed(self, new_file_size: int = 0) -> None:
326        """
327        Проверяет размер кэша и удаляет старые файлы при превышении лимита.
328        
329        Args:
330            new_file_size (int): Размер нового файла, который будет добавлен и который нельзя удалять
331        Returns:
332            None
333        """
334        if not self.cache_enabled or self.cache_max_bytes <= 0:
335            return
336        
337        files_info = self._get_cached_files_info()
338        if not files_info:
339            return
340        
341        # Вычисляем текущий размер кэша
342        current_size = sum(size for _, size, _ in files_info)
343        total_size_with_new = current_size + new_file_size
344
345        # Если размер с новым файлом превышает лимит
346        if total_size_with_new > self.cache_max_bytes:
347            self.logger.warning(f"Размер кэша ({current_size / (1024**2):.2f} MB) превышает лимит "
348                               f"({self.cache_max_bytes / (1024**2):.2f} MB). Удаляем старые файлы...")
349            
350            deleted_count = 0
351            deleted_size = 0
352            
353            # Удаляем файлы, начиная с самых старых
354            for file_path, file_size, mtime in files_info:
355                if total_size_with_new <= self.cache_max_bytes:
356                    break
357                
358                try:
359                    file_path.unlink()
360                    deleted_count += 1
361                    deleted_size += file_size
362                    total_size_with_new -= file_size
363                except Exception as e:
364                    self.logger.error(f"Ошибка при удалении {file_path.name}: {e}")
365            
366            if deleted_count > 0:
367                self.logger.info(f"Удалено файлов: {deleted_count}, освобождено: {deleted_size / (1024**2):.2f} MB")

369    def _handle_response(self, response: requests.Response) -> Dict[str, Any]:
370        """
371        Обрабатывает ответ сервера, извлекая сообщения и проверяя ошибки.

373        Args:

375            
376        Returns:
377            Dict[str, Any]: Данные ответа
378            
379        Raises:
380            requests.exceptions.HTTPError: Если произошла ошибка HTTP
381        """
382        try:
383            data = response.json()
384            self.last_message = data.get("message", "")   
385            return data
386        except ValueError:
387            # Если ответ не JSON (например, файл)
388            self.last_message = response.headers.get("X-Message", "")
389            return {"raw_response": response}

446            self._cleanup_cache_if_needed(0)

488            self.logger.info("Аутентификация успешна")

491            if server_message:
492                self._log_server_message(server_message)
493            
494            # проверка актуальности версии клиента
495            last_client_version = auth_info.get('user', {}).get('version')
496            if last_client_version and self.client_version:
497                # Безопасное сравнение версий
498                try:
499                    current_version_num = int(self.client_version.replace(".", ""))

501                    if current_version_num < last_version_num:
502                        self.logger.warning(f"Версия вашего приложения устарела! Актуальная версия: {last_client_version}. Ваша версия: {self.client_version}")

504                except (ValueError, AttributeError):
505                    pass
506            
507            return auth_info
508        except requests.exceptions.HTTPError as e:
509            self.logger.error(f"Ошибка аутентификации: {e}")
510            if e.response.status_code == 401:
511                try:
512                    error_data = e.response.json()
513                    server_message = error_data.get('message')
514                    if server_message:
515                        self._log_server_message(server_message)
516                except:
517                    pass

 90        if bytes_per_second >= 1024 ** 3:  # GB/s
 91            return f"{bytes_per_second / (1024 ** 3):.2f} GB/s"
 92        elif bytes_per_second >= 1024 ** 2:  # MB/s
 93            return f"{bytes_per_second / (1024 ** 2):.2f} MB/s"
 94        elif bytes_per_second >= 1024:  # KB/s
 95            return f"{bytes_per_second / 1024:.2f} KB/s"
 96        else:  # B/s
 97            return f"{bytes_per_second:.0f} B/s"

 99    def _format_time(self, seconds: float) -> str:
100        """
101        Форматирует время в человеко-читаемый вид.
102        
103        Args:
104            seconds (float): Время в секундах
105            
106        Returns:
107            str: Отформатированная строка времени
108        """
109        if seconds < 1:
110            return f"{seconds * 1000:.0f} мс"
111        elif seconds < 60:
112            return f"{seconds:.1f} с"
113        elif seconds < 3600:
114            minutes = int(seconds // 60)
115            secs = seconds % 60
116            return f"{minutes} мин {secs:.0f} с"
117        else:
118            hours = int(seconds // 3600)
119            minutes = int((seconds % 3600) // 60)
120            secs = seconds % 60
121            return f"{hours} ч {minutes} мин {secs:.0f} с"

263    def _parse_cache_size(self, size_str: str) -> int:
264        """
265        Парсит строку с размером кэша (например, "500 MB", "1GB", "100KB").
266        
267        Args:
268            size_str (str): Строка с размером
269            
270        Returns:
271            int: Размер в байтах
272        """
273        if not size_str or size_str == "0":
274            return 0
275        
276        size_str = size_str.lower().replace(" ", "")
277        
278        # Регулярное выражение для поиска числа и единицы измерения
279        pattern = r'^(\d+(?:\.\d+)?)([kmgt]?b)$'
280        match = re.match(pattern, size_str)
281        
282        if not match:
283            error_message = f"Неверный формат размера кэша в config.yaml: {size_str}. Используйте формат типа '500MB', '1GB', '100KB'"
284            self.logger.error(error_message)
285            sys.exit(1)  
286        
287        size_value = float(match.group(1))
288        unit = match.group(2)
289        
290        units = {
291            'b': 1,
292            'kb': 1024,
293            'mb': 1024 ** 2,
294            'gb': 1024 ** 3,
295            'tb': 1024 ** 4
296        }
297        
298        if unit not in units:
299            error_message = f"Неизвестная единица измерения в config.yaml: {unit}. Используйте B, KB, MB, GB, TB"
300            self.logger.error(error_message)
301            sys.exit(1)       
302        
303        return int(size_value * units[unit])

312        cache_path = Path(self.cache_dir)

316        files_info = []
317        for cache_file in cache_path.glob("*.nxs"):

320                files_info.append((cache_file, stat.st_size, stat.st_mtime))

326    def _cleanup_cache_if_needed(self, new_file_size: int = 0) -> None:
327        """
328        Проверяет размер кэша и удаляет старые файлы при превышении лимита.
329        
330        Args:
331            new_file_size (int): Размер нового файла, который будет добавлен и который нельзя удалять
332        Returns:
333            None
334        """
335        if not self.cache_enabled or self.cache_max_bytes <= 0:
336            return
337        
338        files_info = self._get_cached_files_info()
339        if not files_info:
340            return
341        
342        # Вычисляем текущий размер кэша
343        current_size = sum(size for _, size, _ in files_info)
344        total_size_with_new = current_size + new_file_size
345
346        # Если размер с новым файлом превышает лимит
347        if total_size_with_new > self.cache_max_bytes:
348            self.logger.warning(f"Размер кэша ({current_size / (1024**2):.2f} MB) превышает лимит "
349                               f"({self.cache_max_bytes / (1024**2):.2f} MB). Удаляем старые файлы...")
350            
351            deleted_count = 0
352            deleted_size = 0
353            
354            # Удаляем файлы, начиная с самых старых
355            for file_path, file_size, mtime in files_info:
356                if total_size_with_new <= self.cache_max_bytes:
357                    break
358                
359                try:
360                    file_path.unlink()
361                    deleted_count += 1
362                    deleted_size += file_size
363                    total_size_with_new -= file_size
364                except Exception as e:
365                    self.logger.error(f"Ошибка при удалении {file_path.name}: {e}")
366            
367            if deleted_count > 0:
368                self.logger.info(f"Удалено файлов: {deleted_count}, освобождено: {deleted_size / (1024**2):.2f} MB")
